    rule_ref: "4.3.5"
    text: "Lord without a Siege marker on him; may be Bypassed."

    rule_ref: "1.3.1"

      territory:

        - "Parias Taifa is Neutral."
        - "Independent Taifa is Muslim."

    rule_ref: "4.3.2"
    text: >
      A Lord is Laden if Cart or Mule carries two Provender (incl. shared, 1.5.2),
      or Cart carries any Provender over a Pass, or moving any Loot.
  greed:
    rule_ref: "1.7.2"
    text: >
      Discard Assets only to: March Laden, March Unladen, Avoid Battle, Retreat,
      Feed (Mules excess only, 4.8.1), or per Wastage (4.9.4).
  feed_consumption:
    rule_ref: "4.8.1"
    formula: "ceil((units + mules) / 6)  Provender or Loot per Lord"

          - "Adjust Taifa status (1.4) AFTER placing/removing victory markers"
        no_spoils: true

    siegeworks_check:

      when: "Stronghold did not Surrender (or didn't roll)"
      requires:
        - "besieging_lords_count >= stronghold_capacity"
      effect: "Add 1 Siege marker, max 4"
